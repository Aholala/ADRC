cmake_minimum_required(VERSION 3.16)
project(adrc VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(adrc STATIC main/ALG_ADRC.cpp)
target_include_directories(adrc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/main)
target_compile_features(adrc PUBLIC cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(adrc PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
include(CTest)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    add_link_options(-static)
endif()
if(BUILD_TESTING)
    foreach(name leso2 ladrc1 saturation leso3 ladrc2 ltd2 rleso1 td2 fal eso3 nlesf2 eso2 nlesf1 runtime_options)
        add_executable(${name}_test tests/${name}_test.cpp)
        target_link_libraries(${name}_test PRIVATE adrc)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        endif()
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
endif()
option(ADRC_BUILD_EXAMPLES "Build host teaching runners" OFF)
if(ADRC_BUILD_EXAMPLES)
    add_executable(eso_runner examples/27_eso/eso_runner.cpp)
    add_executable(adrc_example examples/29_adrc/control_example.cpp)
    add_executable(application_runner examples/applications/application_runner.cpp)
    target_link_libraries(eso_runner PRIVATE adrc)
    target_link_libraries(adrc_example PRIVATE adrc)
    target_link_libraries(application_runner PRIVATE adrc)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(eso_runner PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        target_compile_options(adrc_example PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        target_compile_options(application_runner PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
    endif()
endif()
