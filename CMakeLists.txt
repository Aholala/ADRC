cmake_minimum_required(VERSION 3.16)
project(adrc VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(adrc STATIC
    src/fal.cpp src/ltd2.cpp src/nltd2.cpp
    src/leso2.cpp src/leso3.cpp src/rleso1.cpp src/nleso3.cpp
    src/ladrc1.cpp src/ladrc2.cpp src/nlsef2.cpp)
target_include_directories(adrc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(adrc PUBLIC cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(adrc PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
include(CTest)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    add_link_options(-static)
endif()
if(BUILD_TESTING)
    foreach(name leso2 ladrc1 saturation leso3 ladrc2 ltd2 rleso1 nltd2 fal nleso3 nlsef2)
        add_executable(${name}_test tests/${name}_test.cpp)
        target_link_libraries(${name}_test PRIVATE adrc)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        endif()
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
endif()
option(ADRC_BUILD_EXAMPLES "Build host teaching runners" OFF)
if(ADRC_BUILD_EXAMPLES)
    add_executable(nleso_runner examples/27_nleso/eso_runner.cpp)
    add_executable(nladrc_example examples/29_nladrc/control_example.cpp)
    target_link_libraries(nleso_runner PRIVATE adrc)
    target_link_libraries(nladrc_example PRIVATE adrc)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(nleso_runner PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        target_compile_options(nladrc_example PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
    endif()
endif()
