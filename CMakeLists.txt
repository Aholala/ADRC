cmake_minimum_required(VERSION 3.16)
project(adrc VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(adrc STATIC
    src/fal.cpp src/linear_td2.cpp src/nonlinear_td2.cpp
    src/linear_eso.cpp src/linear_eso3.cpp src/reduced_linear_eso.cpp src/nonlinear_eso3.cpp
    src/linear_adrc.cpp src/linear_adrc2.cpp src/nonlinear_feedback2.cpp)
target_include_directories(adrc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(adrc PUBLIC cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(adrc PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
include(CTest)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    add_link_options(-static)
endif()
if(BUILD_TESTING)
    foreach(name linear_eso linear_adrc saturation linear_eso3 linear_adrc2 linear_td2 reduced_linear_eso nonlinear_td2 fal nonlinear_eso3 nonlinear_feedback2)
        add_executable(${name}_test tests/${name}_test.cpp)
        target_link_libraries(${name}_test PRIVATE adrc)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        endif()
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
endif()
option(ADRC_BUILD_EXAMPLES "Build host teaching runners" OFF)
if(ADRC_BUILD_EXAMPLES)
    add_executable(nonlinear_eso_runner examples/27_nonlinear_eso/eso_runner.cpp)
    add_executable(nonlinear_adrc_example examples/29_nonlinear_adrc/control_example.cpp)
    target_link_libraries(nonlinear_eso_runner PRIVATE adrc)
    target_link_libraries(nonlinear_adrc_example PRIVATE adrc)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(nonlinear_eso_runner PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        target_compile_options(nonlinear_adrc_example PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
    endif()
endif()
